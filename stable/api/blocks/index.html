    def append(self, locations: Locations) -> FunctionalBlock:
    """Return this block extended by locations, retaining its operator and name.

    The new locations must have the same input dimension. Supports JIT and
    differentiation; the original block is unchanged.
    """
    if locations.input_dimension != self.locations.input_dimension:
        raise ValueError("appended locations must have the same input dimension")
    return FunctionalBlock(
        Locations(jnp.concatenate((self.locations.points, locations.points))),
        self.operator,
        self.name,
    )

    def __init__(
    self,
    functional: FunctionalBlock,
    variance: PositiveParameter | Scalar | float,
    *,
    max_size: int | None = None,
    count: Count | int | None = None,
) -> None:
    if isinstance(variance, PositiveParameter):
        value = variance.value
    else:
        variance = jnp.asarray(variance)
        variance = error_if(
            variance,
            jnp.any(~jnp.isfinite(variance) | (variance < 0)),
            "noise variances must be finite and nonnegative",
        )
        value = variance
    if value.ndim != 0:
        raise ValueError("noise variance must be scalar")
    supplied = functional.locations.count
    capacity = max(1024, supplied) if max_size is None else max_size
    if isinstance(capacity, bool) or not isinstance(capacity, int):
        raise TypeError("max_size must be an integer")
    if capacity < supplied:
        raise ValueError("max_size must be at least the supplied location count")
    count = jnp.asarray(supplied if count is None else count)
    if count.ndim != 0 or not jnp.issubdtype(count.dtype, jnp.integer):
        raise ValueError("active count must be an integer scalar")
    count = error_if(
        count,
        (count < 0) | (count > supplied),
        "active count must lie within the supplied locations",
    )
    points = jnp.pad(
        functional.locations.points,
        ((0, capacity - supplied), (0, 0)),
        mode="edge",
    )
    points = jnp.where((jnp.arange(capacity) < count)[:, None], points, points[0])
    self.functional = eqx.tree_at(lambda f: f.locations.points, functional, points)
    self.variance = variance
    self.count = count

    def append(self, locations: Locations) -> ObservationDesignBlock:
    """Fill unused locations, retaining operator, name, variance and capacity.

    Locations have shape ``(m, d)`` with the existing input dimension ``d``.
    The returned design supports JIT and differentiation.
    """
    if locations.input_dimension != self.functional.locations.input_dimension:
        raise ValueError("appended locations must have the same input dimension")
    if locations.count > self.capacity:
        raise ValueError("appended locations exceed max_size")
    points, count = error_if(
        (self.functional.locations.points, self.count),
        self.count + locations.count > self.capacity,
        "appended locations exceed max_size",
    )
    points = jax.lax.dynamic_update_slice_in_dim(
        points, locations.points, count, axis=0
    )
    count = count + locations.count
    points = jnp.where(
        (jnp.arange(self.capacity) < count)[:, None], points, points[0]
    )
    return eqx.tree_at(
        lambda b: (b.functional.locations.points, b.count), self, (points, count)
    )

    def replace(self, points: Points) -> ObservationDesignBlock:
    """Replace all active locations, preserving capacity and shared variance.

    Empty contents retain the previous valid padding point. Supports JIT
    and differentiation. Different input shapes specialize this operation,
    but its result always has the original reserved shapes.
    """
    points = jnp.asarray(points)
    expected = self.functional.locations.input_dimension
    if points.ndim != 2 or points.shape[1] != expected:
        raise ValueError(f"replacement locations must have shape (n, {expected})")
    count = points.shape[0]
    if count > self.capacity:
        raise ValueError("replacement locations exceed max_size")
    if count == 0:
        points = self.functional.locations.points[:1]
    functional = FunctionalBlock(
        points, self.functional.operator, self.functional.name
    )
    return ObservationDesignBlock(
        functional, self.variance, max_size=self.capacity, count=count
    )

    def __init__(self, design: ObservationDesignBlock, y: FunctionalValues) -> None:
    y = jnp.asarray(y)
    functional = design.functional
    shape = functional.shape
    if y.ndim != len(shape) or y.shape[1:] != shape[1:] or y.shape[0] > shape[0]:
        raise ValueError(
            f"values for {functional.name!r} have shape {y.shape}; expected {shape}"
        )
    if y.shape[0] < design.capacity:
        y = error_if(
            y,
            design.count != y.shape[0],
            "values must match the active location count",
        )
    y = jnp.pad(y, ((0, design.capacity - y.shape[0]), *((0, 0),) * (y.ndim - 1)))
    self.design = design
    self.y = jnp.where(design.active.reshape((-1,) + (1,) * (y.ndim - 1)), y, 0)

    @classmethod
def from_arrays(
    cls,
    points: Points,
    y: FunctionalValues,
    *,
    variance: PositiveParameter | Scalar | float,
    name: str,
    operator: LinearOperator | None = None,
    max_size: int | None = None,
    padding_point: Point | None = None,
) -> ObservationBlock:
    """Construct observations directly from coordinates and outcomes.

    ``points`` has shape ``(n, d)``. The default identity operator expects
    ``y`` of shape ``(n,)``; otherwise use ``(n, *operator.output_shape(d))``.
    ``variance`` is one scalar measurement-error variance shared by the block.
    Numerical values may be traced; the name and shapes must remain static.

    ``max_size`` reserves locations, defaulting to ``max(1024, n)``.
    Unused slots repeat the first point. For ``n=0``, supply a
    ``padding_point`` valid for the kernel, mean and operator, and an array
    ``points`` with shape ``(0, d)``. Padding is excluded from inference.
    """
    operator = Identity() if operator is None else operator
    points = jnp.asarray(points)
    if points.ndim != 2:
        raise ValueError("locations must have shape (n, d)")
    count = points.shape[0]
    if count == 0:
        if padding_point is None:
            raise ValueError("empty observations require a valid padding_point")
        padding_point = jnp.asarray(padding_point)
        if padding_point.shape != (points.shape[1],):
            raise ValueError("padding_point must match the input dimension")
        points = padding_point[None, :]
    return cls(
        ObservationDesignBlock(
            FunctionalBlock(points, operator, name),
            variance,
            max_size=max_size,
            count=count,
        ),
        y,
    )

    def append(self, locations: Locations, y: FunctionalValues) -> ObservationBlock:
    """Append outcomes using this block's operator and shared variance.

    Locations have shape ``(m, d)`` with the existing input dimension ``d``;
    ``y`` must have shape ``(m, *output_shape)``. Returns a new block and
    supports JIT and differentiation, including the shared variance parameter.
    """
    design = self.design.append(locations)
    y = jnp.asarray(y)
    expected_shape = (locations.count, *self.functional.shape[1:])
    if y.shape != expected_shape:
        raise ValueError(
            f"appended values have shape {y.shape}; expected {expected_shape}"
        )
    values = jax.lax.dynamic_update_slice_in_dim(self.y, y, self.count, axis=0)
    return ObservationBlock(design, values)

    @classmethod
def from_blocks(
    cls,
    blocks: Sequence[FunctionalBlock],
) -> IndexMap:
    indices: list[BlockIndex] = []
    start = 0
    for block in blocks:
        stop = start + block.size
        indices.append(BlockIndex(block.name, block.shape, start, stop))
        start = stop
    return cls(tuple(indices), start)

    leading_shape = flat.shape[:-1]
    return {
        block.name: flat[..., block.slice].reshape((*leading_shape, *block.shape))
        for block in self.blocks
    }

    def pack(self, values: Mapping[str, FunctionalValues]) -> FlatValues:
    extra = set(values) - {block.name for block in self.blocks}
    if extra:
        raise ValueError(f"unexpected blocks: {sorted(extra)}")
    parts = []
    for block in self.blocks:
        value = jnp.asarray(values[block.name])
        if value.shape != block.shape:
            raise ValueError(
                f"block {block.name!r} has shape {value.shape}; "
                f"expected {block.shape}"
            )
        parts.append(value.reshape(-1))
    return jnp.concatenate(parts)

    ObservationDesignBlock.variance stores the scalar measurement-error variance. Use a fixed scalar or a scalar PositiveParameter to fit it. The same variance applies to every location and output component in that block.

    The block's variances property evaluates that parameter and repeats it over the functional output shape. Collections expose these measurement-error variances in canonical scalar-output order through variance_flat.

    Observation designs reserve max_size locations and keep a dynamic active count. See JAX and observation capacity for capacity, padding and compilation behavior. Functional queries are unreserved.
